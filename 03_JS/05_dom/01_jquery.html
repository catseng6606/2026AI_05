      <script>
        // 當網頁 DOM 載入完成後執行
        $(document).ready(function() {
          
          // ==========================================
          // 1. ID 選擇器對照
          // ==========================================
          // jQuery: 使用 $() 選取 ID 並使用 .text() 賦值
          $('#jq-id-demo').text('【jQuery】成功使用 ID 選擇器更改文字！');

          // Native JS: 使用 getElementById() 並使用 .innerText 屬性賦值
          document.getElementById('js-id-demo').innerText = '【Native JS】使用 getElementById 更改文字！';

          // Native JS (類似 jQuery 寫法): 使用 querySelector() 並帶入 CSS 選擇器 \'#id\'
          document.querySelector('#js-query-demo').innerText = '【Native JS】使用 querySelector(\\\'#id\\\') 更改文字！';

          // ==========================================
          // 2. Class 選擇器對照
          // ==========================================
          // jQuery: 會自動將選取到的所有 class 元素一次變更 (隱式迭代)
          $('.jq-class-demo').text('【jQuery】成功使用 Class 選擇器更改所有元素的文字！');

          // Native JS: getElementsByClassName 回傳的是類陣列，必須透過迴圈逐一更改元素
          var jsClassElements = document.getElementsByClassName('js-class-demo');
          for (var i = 0; i < jsClassElements.length; i++) {
            jsClassElements[i].innerText = '【Native JS】使用 Class 與迴圈更改文字 (第 ' + (i + 1) + ' 個)';
          }

          // ==========================================
          // 3. 標籤選擇器對照
          // ==========================================
          // jQuery: 選取 #jq-tag-container span 並用 .text() 一次變更
          $('#jq-tag-container span').text('【jQuery】標籤選擇器變更');

          // Native JS: 使用 querySelectorAll 選取後，用 forEach 迴圈逐一變更
          var jsTagElements = document.querySelectorAll('#js-tag-container span');
          jsTagElements.forEach(function(element, index) {
            element.innerText = '【Native JS】標籤選擇器變更 (第 ' + (index + 1) + ' 個)';
          });

        });
      </script>
