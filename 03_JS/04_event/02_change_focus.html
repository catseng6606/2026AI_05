JavaScript 事件學習：表單事件 & 滑鼠事件

事件觀念說明：
  • onfocus：當輸入框「取得焦點」（點擊輸入框內）時觸發。
  • onblur：當輸入框「失去焦點」（點擊輸入框外）時觸發。（註：沒有 onblue 事件，正確名稱為 onblur）
  • onchange：當輸入框「內容被修改」且「失去焦點」後，內容與原先不同時觸發。
  • onmouseover：當滑鼠游標「移入」元素上方時觸發。
  • onmouseout：當滑鼠游標「移出」元素上方時觸發。
實體元件：表單事件測試區
onchange 接收區：尚未偵測到內容變更。
表單事件狀態：等待操作中...
實體元件：滑鼠事件測試區

教學範例原始碼

<!-- 1. 表單事件 HTML 結構 -->
<input
  type="text"
  id="myInput"
  class="form-control"
  onfocus="handleFocus()"
  onblur="handleBlur()"
  onchange="handleChange()"
/>

<div id="changeResultDiv" class="p-3 mb-3 border rounded bg-light text-muted">
  onchange 接收區：尚未偵測到內容變更。
</div>

<!-- 2. 滑鼠事件 HTML 結構 -->
<div
  id="mouseArea"
  class="p-5 border rounded text-center bg-light text-muted fw-bold"
  onmouseover="handleMouseOver()"
  onmouseout="handleMouseOut()"
>
  滑鼠尚未移入
</div>

<script>
  // === 表單事件處理 ===
  function handleFocus() {
    document.getElementById("statusMsg").innerText = "觸發了 onfocus (取得焦點)";
    updateStatusClass("alert-primary");
  }

  function handleBlur() {
    document.getElementById("statusMsg").innerText = "觸發了 onblur (失去焦點)";
    updateStatusClass("alert-danger");
  }

  function handleChange() {
    document.getElementById("statusMsg").innerText = "觸發了 onchange (內容已改變)";
    updateStatusClass("alert-success");

    const inputValue = document.getElementById("myInput").value;
    const resultDiv = document.getElementById("changeResultDiv");
    resultDiv.innerText = "onchange 接收區：已更新內容為「" + inputValue + "」";
    resultDiv.className = "p-3 mb-3 border rounded bg-light text-dark fw-bold";
  }

  function updateStatusClass(newClass) {
    const alertDiv = document.getElementById("statusAlert");
    alertDiv.className = "alert mb-0 " + newClass;
  }

  // === 滑鼠事件處理 ===
  function handleMouseOver() {
    const mouseDiv = document.getElementById("mouseArea");
    mouseDiv.innerText = "觸發了 onmouseover (滑鼠已移入！)";
    mouseDiv.className = "p-5 border rounded text-center bg-warning text-dark fw-bold shadow-sm";
  }

  function handleMouseOut() {
    const mouseDiv = document.getElementById("mouseArea");
    mouseDiv.innerText = "觸發了 onmouseout (滑鼠已移出！)";
    mouseDiv.className = "p-5 border rounded text-center bg-light text-muted fw-bold";
  }
</script>
