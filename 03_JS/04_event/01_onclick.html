JavaScript 按鈕點擊事件 (onclick) 示範

教學原始碼對照

<!-- 有 onclick 屬性的按鈕 -->
<button class="btn btn-primary" onclick="showAlert()">有 onclick 的按鈕 (行內事件)</button>

<!-- 沒有 onclick 屬性的按鈕 -->
<button id="btnNoOnclick" class="btn btn-secondary">沒有 onclick 的按鈕</button>

<script>
  // 行內事件所觸發的函式
  function showAlert() {
    alert('您點擊了有 onclick 屬性的按鈕！');
  }
</script>
