let gold = 50;
let hasKey = false;

// 條件判斷：金幣滿80「或」有鑰匙
if ( gold >= 80 || hasKey === true ) {
// 條件成立執行這區塊 (開門)
$('#door-visual').css('border-color', '#10b981');
$('#door-icon').removeClass('fa-door-closed text-slate-500')
.addClass('fa-door-open text-emerald-400');

} else {
// 條件不成立執行這區塊 (拒絕)
$('#door-visual').css('border-color', '#ef4444');
$('#door-icon').removeClass('fa-door-open text-emerald-400')
.addClass('fa-door-closed text-red-500');
$('#door-status').text('條件不足，無法進入').css('color', '#f87171');
}

$('#star-container').empty(); // 先清空容器
let count = 5;

// 宣告變數 i=0; 若 i < count 繼續執行; 每次結束 i+1
for (let i = 0; i < count; i++) {
let starHTML = '<i class="fa-solid fa-star text-yellow-400"></i>';

// 迴圈每跑一次，就向容器中添加一個星星
$('#star-container').append(starHTML);
}
let power = 0; // 初始電量
let step = 20; // 每次充電量

// 當(while)電量小於 100 時，持續執行區塊內程式
while (power < 100) {
power = power + step; // 增加電量

// 確保不超過 100
if (power > 100) power = 100;

// 更新畫面 (這行在真實網頁會瞬間完成)
// 我們用動畫延遲來讓你看見過程！
$('#battery-level').css('width', power + '%');
}
// jQuery 特有！選取多個元素後，對「每一個」執行函式
$('.soldier-card').each(function( index, element ) {

// $(this) 代表當前正在處理的這個元素
$(this).addClass('bg-blue-600 border-blue-400 text-white');
$(this).removeClass('bg-slate-700 text-slate-400');

// index 是從 0 開始的索引值
let number = index + 1;
$(this).find('.s-num').text('No.' + number);

});
